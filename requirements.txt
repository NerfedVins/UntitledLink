# yt-dlp is deliberately unpinned: it breaks whenever a site changes its
# markup, and the app refreshes it in the background on every launch. The rest
# carry floors rather than pins - the APIs used here are old and stable, but a
# machine with a five-year-old Pillow should be told rather than fail oddly.
yt-dlp
# The [socks] extra is PySocks, and without it a socks5 proxy - which is
# what Tor is - fails at the first request with "Missing dependencies for
# SOCKS support". yt-dlp carries its own SOCKS code, so leaving it out
# broke half the app and left the other half working.
requests[socks]>=2.31
beautifulsoup4>=4.12
pillow>=10.0

# ffmpeg, as a wheel. This is the same gyan.dev static build the app used to
# download for itself at runtime, except pip checks its hash on the way in and
# there is no pinned digest here to go stale. Needed for 1080p and above (those
# ship video and audio separately and have to be merged) and for mp3.
imageio-ffmpeg>=0.5
